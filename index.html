<script>

(() => {

  const reduce = window.matchMedia('(prefers-reduced-motion: reduce)');

  const desktop = window.matchMedia('(min-width: 900px)');

  const root = document.documentElement;

  const nav = document.querySelector('.nav');

  const hero = document.querySelector('.hero');

  const stack = document.querySelector('.stack');

  const layers = [...stack.querySelectorAll('.layer')];

  const labels = layers.map(layer => layer.querySelector('.layer-label'));

  const hint = document.querySelector('.hint');

  const portrait = document.querySelector('.portrait');


  const clamp = (v, lo = 0, hi = 1) => Math.min(hi, Math.max(lo, v));

  const ease = t => (t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2);


  let plateH = 0, gapMin = 0, gapMax = 0, stackH = 0, queued = false;


  // top layer paints above the ones beneath it

  layers.forEach((layer, i) => { layer.style.zIndex = String(layers.length - i); });


  function measure() {

    plateH = layers[0].querySelector('svg').getBoundingClientRect().height;

    gapMin = plateH * 0.07;

    gapMax = plateH * (desktop.matches ? 0.46 : 0.42);

    stackH = plateH + gapMax * (layers.length - 1);

    stack.style.height = stackH + 'px';

  }


  // 0 = layers stacked together, 1 = fully separated

  function heroProgress() {

    if (reduce.matches) return 1;

    const vh = window.innerHeight;

    if (desktop.matches) {

      const travel = hero.offsetHeight - vh;

      return travel > 0 ? clamp((-hero.getBoundingClientRect().top / travel) * 1.2) : 1;

    }

    return clamp(window.scrollY / (vh * 0.55));

  }


  function render() {

    queued = false;

    const p = ease(heroProgress());

    const mid = (layers.length - 1) / 2;

    const gap = gapMin + (gapMax - gapMin) * p;

    const base = (stackH - plateH) / 2;


    layers.forEach((layer, i) => {

      layer.style.transform = 'translate3d(0,' + (base + (i - mid) * gap).toFixed(2) + 'px,0)';

      labels[i].style.opacity = clamp((p - 0.2 - i * 0.1) / 0.25).toFixed(3);

    });

    if (hint) hint.style.opacity = (1 - clamp(p * 4)).toFixed(3);

    nav.classList.toggle('is-scrolled', window.scrollY > 8);


    if (reduce.matches) return;


    // background drifts at a fifth of scroll speed

    root.style.setProperty('--bgy', (-window.scrollY * 0.2).toFixed(1) + 'px');


    // portrait moves slower than its frame

    const img = portrait && portrait.querySelector('img');

    if (img) {

      const r = portrait.getBoundingClientRect();

      const vh = window.innerHeight;

      if (r.bottom > 0 && r.top < vh) {

        const t = clamp((vh - r.top) / (vh + r.height));

        img.style.transform = 'translate3d(0,' + ((t - 0.5) * 14).toFixed(2) + '%,0)';

      }

    }

  }


  function queue() {

    if (!queued) { queued = true; requestAnimationFrame(render); }

  }


  function refresh() { measure(); render(); }


  window.addEventListener('scroll', queue, { passive: true });

  window.addEventListener('resize', refresh);

  window.addEventListener('load', refresh);

  desktop.addEventListener('change', refresh);

  reduce.addEventListener('change', refresh);

  refresh();


  document.getElementById('year').textContent = new Date().getFullYear();

})();
